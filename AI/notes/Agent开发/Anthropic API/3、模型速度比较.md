这段函数用于对同一个第三方网关下的多个模型发送相同提示词，然后统计每个模型的响应内容、输出 token 数、总耗时以及平均每个输出 token 的生成耗时。

```python
import token  
  
from dotenv import load_dotenv  
from anthropic import Anthropic  
import os  
import time  
  
load_dotenv()  
  
api_key = os.getenv("ANTHROPIC_API_KEY") or os.getenv("ANTHROPIC_AUTH_TOKEN")  
if not api_key:  
    raise RuntimeError("请设置 ANTHROPIC_AUTH_TOKEN 或 ANTHROPIC_API_KEY 环境变量")  
  
client = Anthropic(  
    base_url=os.getenv("ANTHROPIC_BASE_URL"),  
    api_key=api_key,  
)  
  
def compare_model_speeds():  
    models = [  
          "MiniMax-M3",  
          "Ali-DeepSeek-V4-Pro",  
          "Ali-DeepSeek-V4-Flash",  
          "Baidu-DeepSeek-V4-Pro",  
          "Baidu-DeepSeek-V4-Flash",  
          "Kimi-K2.6",  
          "Ali-GLM-5.1",  
          "GLM-5.1",  
          "Ali-GLM-5",  
          "Doubao-Seed-2.0-pro"  
    ]  
  
    task = "用一个简单的段落解释下深度学习的概念"  
  
    for model in models:  
        # 记录开始时间  
        start_time = time.time()  
  
        response = client.messages.create(  
            model=model,  
            max_tokens=500,  
            messages=[  
                {"role": "user", "content": task}  
            ]  
        )  
  
        end_time = time.time()  
  
        execution_time = end_time - start_time  
  
        tokens = response.usage.output_tokens  
  
        time_per_token = execution_time / tokens  
  
        print(f"Model: {model}")  
        print(f"Response: {response.content[0].text}")  
        print(f"Generated Tokens: {tokens}")  
        print(f"Execution Time: {execution_time: .2f} seconds")  
        print(f"Time per Token: {time_per_token: .2f} seconds")  
  
        print()  
  
  
compare_model_speeds()
```
